#+begin_src elisp
  (defun list-from-table (key table &optional default)
    (if-let ((items (alist-get key table)))
        items
      default))

  (defun value-from-table (key table &optional default)
    (if-let ((value (car (alist-get key table))))
        (float value)
      default))

  (defun soc-alloc-3 (power bounds cids socs sops prev-allocs)
    (let ((pending-power power)
          (pending-lower-bound (car bounds))
          (pending-upper-bound (cadr bounds))
          (target-allocs nil)
          (target-bounds nil)
          (total-soc 0.0))
      (dolist (cid cids)
        (setq total-soc (+ total-soc (- 100.0 (value-from-table cid socs 0.0)))))
      (dolist (cid cids)
        (let* ((soc          (- 100 (value-from-table cid socs 0.0)))
               (sop          (list-from-table cid sops '(0.0 0.0)))
               (sop-lower (car  sop))
               (sop-upper (cadr sop))

               (upper-bound  (min (truncate (* (/ pending-upper-bound total-soc) soc))
                                  sop-upper))
               (lower-bound  (max (truncate (* (/ pending-lower-bound total-soc) soc))
                                  sop-lower))

               (prev-alloc   (value-from-table cid prev-allocs 0.0))
               (raw-alloc    (truncate (* (/ pending-power total-soc) soc)))
               (adj-alloc    (+ prev-alloc (if (< raw-alloc 0)
                                 (max lower-bound raw-alloc)
                               (min upper-bound raw-alloc)))))

            (message
             (format "cid: %s, soc: %s, sop: %s, prev-alloc: %s, raw-alloc: %s, adj-alloc: %s"
                     cid soc sop prev-alloc raw-alloc adj-alloc))
            (message "pending-power: %s, pending-bounds: %s"
                     pending-power
                     (list pending-lower-bound pending-upper-bound))
            (message
             (format "sop: %s; bounds: %s\n" sop (list lower-bound upper-bound)))
          (if (< upper-bound lower-bound)
              (message "not processing")
            (setq pending-power
                  (- pending-power
                     (if (< raw-alloc 0)
                         (max (- lower-bound prev-alloc) raw-alloc)
                       (min (- upper-bound prev-alloc) raw-alloc))))

            (let ((pending-upper (- pending-upper-bound upper-bound))
                  (pending-lower (- pending-lower-bound lower-bound)))
              (message "Adjusting pending bounds: %s, %s" pending-lower pending-upper))

            (setq pending-upper-bound (- pending-upper-bound upper-bound))
            (setq pending-lower-bound (- pending-lower-bound lower-bound))
            (message "Adjusting pending bounds: %s, %s" pending-lower-bound pending-upper-bound)
            
            (push (list cid (- lower-bound adj-alloc) (- upper-bound adj-alloc)) sops)
            (push `(,cid ,adj-alloc) target-allocs))
          (setq total-soc (- total-soc soc))))
      (list target-allocs sops)))
#+end_src
